            <p id="palabraVotar" class="text-3xl text-indigo-400 font-semibold mb-6"></p>

            <div class="flex justify-center gap-4 mb-6">

                <button onclick="votar(1)" class="w-12 h-12 bg-slate-700 hover:bg-indigo-600 rounded-full text-xl font-bold transition">1</button>

                <button onclick="votar(5)" class="w-12 h-12 bg-slate-700 hover:bg-indigo-600 rounded-full text-xl font-bold transition">5</button>

    <script>

        // CONFIGURACIÓN DE SUPABASE (Sustituye con tus datos)

        const SUPABASE_URL = 'TU_SUPABASE_URL';

        const SUPABASE_ANON_KEY = 'TU_SUPABASE_ANON_KEY';

        const supabaseClient = supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);


        // LISTA DE PALABRAS DE TU CURSO (Puedes añadir todas las que quieras)

        const listaPalabras = [

            { fecha: "2026-11-09", palabra: "Inefable", definicion: "Que no se puede explicar con palabras." },

            { fecha: "2026-11-10", palabra: "Melifluo", definicion: "Sonido excesivamente dulce, suave o delicado." },

            { fecha: "2026-11-11", palabra: "Alba", definicion: "Primera luz del día antes de salir el sol." }

        ];


        const urlParams = new URLSearchParams(window.location.search);

        const palabraParam = urlParams.get('palabra');


        if (palabraParam) {

            // MODO MÓVIL (Alumno)

            document.getElementById('seccion-palabra').classList.add('hidden');

            document.getElementById('seccion-votacion').classList.remove('hidden');

            document.getElementById('palabraVotar').innerText = palabraParam;

        } else {

            // MODO PROYECTOR (Clase)

            // Coger la palabra de hoy (o la primera si no coincide ninguna)

            const hoy = new Date().toISOString().split('T')[0];

            let actual = listaPalabras.find(p => p.fecha === hoy) || listaPalabras[0];


            document.getElementById('palabraDia').innerText = actual.palabra;

            document.getElementById('definicionDia').innerText = `"${actual.definicion}"`;


            // Generar QR que apunta a esta misma página pero con parámetro de la palabra

            const linkVotacion = `${window.location.origin}${window.location.pathname}?palabra=${encodeURIComponent(actual.palabra)}`;

            new QRCode(document.getElementById("qrcode"), {

                text: linkVotacion,

                width: 150,

                height: 150

            });

        }


        async function votar(puntuacion) {

            const curso = urlParams.get('curso') || '1ESO';

            const anio = urlParams.get('anio') || '2026-2027';

            

            const { error } = await supabaseClient

                .from('votos_palabras')

                .insert([{ palabra: palabraParam, curso: curso, anio: anio, puntuacion: puntuacion }]);


            if (!error) {

                document.querySelector('.flex.justify-center.gap-4').classList.add('hidden');

                document.getElementById('mensajeVoto').classList.remove('hidden');

            } else {

                alert('Hubo un error al registrar el voto.');

            }

        }


        // Ajustar enlaces del QR si el profesor cambia los desplegables de curso/año en clase

        document.getElementById('selectCurso').addEventListener('change', actualizarQR);

        document.getElementById('selectAnio').addEventListener('change', actualizarQR);


        function actualizarQR() {

            if (palabraParam) return;

            const curso = document.getElementById('selectCurso').value;

            const anio = document.getElementById('selectAnio').value;

            const hoy = new Date().toISOString().split('T')[0];

            let actual = listaPalabras.find(p => p.fecha === hoy) || listaPalabras[0];


            document.getElementById('qrcode').innerHTML = "";

            const linkVotacion = `${window.location.origin}${window.location.pathname}?palabra=${encodeURIComponent(actual.palabra)}&curso=${curso}&anio=${anio}`;

            new QRCode(document.getElementById("qrcode"), {

                text: linkVotacion,

                width: 150,

                height: 150

            });

        }

        // Inicializar QR con los valores por defecto

        actualizarQR();

    </script>
